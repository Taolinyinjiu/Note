  // 门限滤波，对于某些里程计，当有轻微抖动时会产生一个突破当前物理特性的尖峰，尤其是baton
  // mini，因此需要对数据做一个门限滤波，参数先取15吧
  double dist = (msg_pos - pos_history).norm();
  if (dist > 1.0) {
    ROS_WARN("odom error");
    odom_ready= false;
  }
  // 通过门限滤波，做一次低通滤波
  double alpha = 0.9;
  // 更新历史值，对于控制器，我们使用odom的原始值，当然也可以使用0.9参数下的历史值，可能会降低一点精度，但是输入的里程计曲线会光滑很多
  pos_history = (1.0 - alpha) * pos_history + alpha * msg_pos;


	  // 使用滑动平均或者累积平均来估计频率
  double current_hz = 1.0 / dt;
	estimate_hz = 0.95 * estimate_hz + 0.05 * current_hz;
  // 跌落保护
  static uint8_t warn_counter = 0;
  if (estimate_hz < (odom_hz_must * 0.5)) {
		ROS_ERROR_THROTTLE(1.0, "odom hz low");
    if (warn_counter < 10)
      warn_counter++;
    if (warn_counter++ > 5)
      odom_ready= false;
  } else {
    warn_counter = 0;
  }
	odom_ready = true;



	/*
设置控制器的当前位置，通过里程计输入，不同的里程计maybe需要实现不同的tf转换从而得到标准的里程计信息，需要满足三个基本的要求
1. xyz位置
2. xyz速度
3. roll pitch yaw三轴姿态(四元数
*/
void LinerController::set_current_state(const nav_msgs::Odometry::ConstPtr &msg,
                                        uint8_t odom_type) {
  // 记录当前时间戳
  ros::Time now_time = ros::Time::now();                                                                                                 
  // 如果这是第一次进入到该回调函数，那么将该次里程计数据作为基准值,结束函数
  if (last_odom_time_.isZero()) {
    odom_buffer.push_back(*msg); // 基准值
    last_odom_time_ = ros::Time::now();
    odom_ready= false;
		return;
  }
  // 通过两帧时间差估计频率
  double dt = (now_time - last_odom_time_).toSec();
  // 容错处理：过滤掉极小值（防止重复包）或负值
  if (dt <= 0.001){
    odom_ready= false;
		return;
	}

  switch (odom_type) {
  /* ODOM = 0, POSE = 1, GAZEBO = 2,  MOCAP = 3, VIOBOT = 4, GPS = 5, RTK = 6,
   * TODO: 测试阶段需要写的有:0 ODOM 2 GAZEBO 3 MOCAP ４ VIOBOT */
  case 0: { // 标准里程计话题，直接输入
    // 位置
    current_state.pos[0] = msg->pose.pose.position.x;
    current_state.pos[1] = msg->pose.pose.position.y;
    current_state.pos[2] = msg->pose.pose.position.z;
    // 速度
    current_state.vel[0] = msg->twist.twist.linear.x;
    current_state.vel[1] = msg->twist.twist.linear.y;
    current_state.vel[2] = msg->twist.twist.linear.z;
    // 姿态
    current_state.q.x() = msg->pose.pose.orientation.x;
    current_state.q.y() = msg->pose.pose.orientation.y;
    current_state.q.z() = msg->pose.pose.orientation.z;
    current_state.q.w() = msg->pose.pose.orientation.w;
    // 偏航
    current_state.yaw =
        get_yaw_from_quaternion(current_state.q);
    last_odom_time_ = now_time;
    odom_ready= true;
    break;
  }
  case 1: { // 使用POSE话题传入的里程计我还没遇到过，先放着吧
    break;
  }
  case 2: { // GAZEBO里程计，仿真使用，控制器设计前期比较重要,external_fusion直接用的上面里程计的回调函数给Gazebo用，就改了下订阅名o.O
    break;
  }
  case 3: { // 动捕，实机测试使用，并且期望在使用vio或者lio时，如果他们的里程计漂移了可以使用动捕的救回来
    // 动捕的话题有个缺点就是他的位置和速度是分开发的，因此外部需要重订阅一次动捕话题，然后输出到一个话题中，以实现位置数据同步
    current_state.pos[0] = msg->pose.pose.position.x;
    current_state.pos[1] = msg->pose.pose.position.y;
    current_state.pos[2] = msg->pose.pose.position.z;
    // 速度
    current_state.vel[0] = msg->twist.twist.linear.x;
    current_state.vel[1] = msg->twist.twist.linear.y;
    current_state.vel[2] = msg->twist.twist.linear.z;
    // 姿态
    current_state.q.x() = msg->pose.pose.orientation.x;
    current_state.q.y() = msg->pose.pose.orientation.y;
    current_state.q.z() = msg->pose.pose.orientation.z;
    current_state.q.w() = msg->pose.pose.orientation.w;
    // 偏航
    current_state.yaw =
        get_yaw_from_quaternion(current_state.q);
    last_odom_time_ = now_time;
    odom_ready= true;
    break;
  }
  case 4: { // viobot的里程计
            // 1.
    // 首先,viobot输出的里程计的position是已经转换过的,也就是说可以直接使用,因此
    current_state.pos(0) = msg->pose.pose.position.x;
    current_state.pos(1) = msg->pose.pose.position.y;
    current_state.pos(2) = msg->pose.pose.position.z;
    // 2. 姿态部分viobot的姿态是未经转换的
    /*         定义从相机原始坐标系到 FLU (Forward-Left-Up) 的旋转转换
        假设相机安装方式：镜头向前，但内部坐标系为 Optical (Z向前, X向右, Y向下)
        绕 Z 轴旋转 90°，再绕 Y 轴旋转 -90°
        通过 q_z(90°) 和 q_y(-90°) 将其对齐为 FLU (X向前, Y向左, Z向上)
    */
    tf2::Quaternion q_z, q_y;
    q_z.setRPY(0, 0, M_PI / 2);
    q_y.setRPY(0, -M_PI / 2, 0);

    tf2::Quaternion q_cam_to_flu = q_z * q_y;
    tf2::Matrix3x3 q_rot_matrix(q_cam_to_flu);
    //  处理姿态转换(四元数)
    tf2::Quaternion q_orig(
        msg->pose.pose.orientation.x, msg->pose.pose.orientation.y,
        msg->pose.pose.orientation.z, msg->pose.pose.orientation.w);
    // 将原始姿态转到 FLU 框架下
    tf2::Quaternion q_final = q_orig * q_cam_to_flu;
    // 将转换厚的四元数变换为欧拉角
    tf2::Matrix3x3 m(q_final);
    double roll, pitch, yaw;
    m.getRPY(roll, pitch, yaw);
    // 填充数据
    current_state.q.w() = static_cast<float>(q_final.w());
    current_state.q.x() = static_cast<float>(q_final.x());
    current_state.q.y() = static_cast<float>(q_final.y());
    current_state.q.z() = static_cast<float>(q_final.z());
    current_state.yaw =
        get_yaw_from_quaternion(current_state.q);

    double orig_vx = msg->twist.twist.linear.x;
    double orig_vy = msg->twist.twist.linear.y;
    double orig_vz = msg->twist.twist.linear.z;

    current_state.vel(0) = static_cast<float>(-orig_vx);
    current_state.vel(1) = static_cast<float>(orig_vy);
    current_state.vel(2) = static_cast<float>(-orig_vz);
    last_odom_time_ = now_time;
    odom_ready= true;
    break;
  }
  default: {
    odom_ready= false;
    break;
  }
  }
	// 在所有处理结束后更新时间戳
  last_odom_time_ = now_time;
}

void LinerController::debug_odom_pub_fusion(const ros::TimerEvent& event) {
    nav_msgs::Odometry odom_msg;
    odom_msg.header.stamp = ros::Time::now();
    odom_msg.header.frame_id = "world";
    odom_msg.child_frame_id = "base_link";
    // 填充位置
    odom_msg.pose.pose.position.x = current_state.pos[0];
    odom_msg.pose.pose.position.y = current_state.pos[1];
    odom_msg.pose.pose.position.z = current_state.pos[2];
		// 填充速度
		odom_msg.twist.twist.linear.x = current_state.vel[0];
		odom_msg.twist.twist.linear.x = current_state.vel[0];
		odom_msg.twist.twist.linear.x = current_state.vel[0];
		// 填充姿态
    odom_msg.pose.pose.orientation.x = current_state.q.x();
    odom_msg.pose.pose.orientation.y = current_state.q.y();
    odom_msg.pose.pose.orientation.z = current_state.q.z();
    odom_msg.pose.pose.orientation.w = current_state.q.w();

    debug_odom_pub_.publish(odom_msg);
}
